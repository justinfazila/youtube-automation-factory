## 🔧 SCRIPT PYTHON POUR GÉNÉRATION (Edge-TTS Local)

```python
# generate_voiceover.py
import edge_tts
import asyncio

VOICE = "fr-FR-HenriNeural"
RATE = "-5%"
PITCH = "+2Hz"
TEXT = """[Insérer le texte brut du script sans balises SSML]"""

async def main():
    communicate = edge_tts.Communicate(TEXT, VOICE, rate=RATE, pitch=PITCH)
    await communicate.save("assets/audio/video-1/voiceover_raw.mp3")

if __name__ == "__main__":
    asyncio.run(main())
```
